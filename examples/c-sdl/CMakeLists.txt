cmake_minimum_required(VERSION 3.28)
project(tilewright_ui_c_sdl_demo LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(SDL3 3.4.18 EXACT CONFIG REQUIRED)
set(TILEWRIGHT_UI_LIBRARY "" CACHE FILEPATH "Built libtilewright_ui_ffi shared library")
if(NOT EXISTS "${TILEWRIGHT_UI_LIBRARY}")
  message(FATAL_ERROR "Set TILEWRIGHT_UI_LIBRARY to the task-built shared library")
endif()
add_executable(tilewright-ui-c-sdl main.c)
target_include_directories(tilewright-ui-c-sdl PRIVATE ../../include)
target_compile_options(tilewright-ui-c-sdl PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(tilewright-ui-c-sdl PRIVATE SDL3::SDL3 "${TILEWRIGHT_UI_LIBRARY}")
enable_testing()
add_executable(tilewright-ui-c-input-test input_test.c)
target_include_directories(tilewright-ui-c-input-test PRIVATE ../../include)
target_compile_options(tilewright-ui-c-input-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(tilewright-ui-c-input-test PRIVATE SDL3::SDL3 "${TILEWRIGHT_UI_LIBRARY}")
add_test(NAME c-host-input COMMAND tilewright-ui-c-input-test)
set_tests_properties(c-host-input PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
